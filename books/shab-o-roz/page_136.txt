# Print typo corrected: "اُسے چھوڑ ا نہیں گیا" -> "اُسے چھوڑا نہیں گیا"; "ری سے ماخوذ" -> "رِی سے ماخوذ".

[ur]
غیر مکفی، یعنی وہ کسی کا محتاج نہیں۔ وہی اپنے بندوں کو کھلاتا ہے اور وہی اُن کے لیے کافی ہے۔ ولا مودع، یعنی اُسے چھوڑا نہیں گیا۔ کفانا، یہ کفایت سے ماخوذ ہے۔ اَرْوانا، یہ رِی سے ماخوذ ہے یعنی سیراب کرنا۔ ولا مکفور، یعنی اُس کی فضیلت اور اُس کے انعامات کا انکار نہیں کیا جا سکتا۔

[h]
اکٹھے مل کر کھانا کھانا

[ur]
اکٹھے مل کر کھانا کھانا سنت ہے۔ اہلِ خانہ کی کوشش یہی ہونی چاہیے کہ سب مل کر کھانا کھائیں۔ بجائے اس کے کہ ہر کوئی اپنا کھانا الگ الگ لے کر کھائے۔

[ur]
حضرت جابر رضی اللہ عنہ سے روایت ہے کہ میں نے رسول اللہ صلی اللہ علیہ وآلہ وسلم کو فرماتے ہوئے سنا: “ایک آدمی کا کھانا دو آدمیوں کے لیے کافی ہے۔ دو آدمیوں کا کھانا چار آدمیوں کے لیے کافی ہے۔ اور چار آدمیوں کا کھانا آٹھ آدمیوں کے لیے کافی ہے۔”

[ref]
(صحیح مسلم، حدیث: 2059۔)

[h]
کھانا پسند آئے تو اُس کی تعریف کرنا

[ur]
کھانا پسند آئے تو اُس کی تعریف کرنا سنت ہے۔

[ur]
حضرت جابر رضی اللہ عنہ سے روایت ہے کہ نبی صلی اللہ علیہ وآلہ وسلم نے اپنے اہلِ خانہ سے سالن مانگا۔ انھوں نے کہا: ہمارے پاس تو صرف سرکہ ہے۔ آپ نے سرکہ ہی منگوا لیا، اُسے کھانے لگے۔ پھر فرمایا: “سرکہ اچھا سالن ہے۔ سرکہ اچھا سالن ہے۔”

[ref]
(صحیح مسلم، حدیث: 2052۔)

[ur]
سرکہ عربوں کے ہاں سالن کے طور پر مستعمل تھا۔ اُن کا سرکہ میٹھا ہوتا تھا۔ آج کل کے سرکے کی طرح کھٹا نہیں ہوتا تھا۔

[ur]
شیخ ابن عثیمین رحمہ اللہ اس کی شرح کرتے ہوئے لکھتے ہیں: یہ بھی نبی صلی اللہ علیہ وآلہ وسلم کی ایک سنت ہے کہ اگر آپ کو کھانا اچھا لگتا تو آپ اُس کی تعریف کرتے تھے۔ مثال کے طور پر تم روٹی کی تعریف کرو تو یوں کہہ سکتے ہو: “بنو فلاں کی روٹی بہت اچھی ہوتی ہے۔” یا اسی طرح کے دیگر جملے کہے جائیں۔ یہ رسول اللہ صلی اللہ علیہ وآلہ وسلم کی سنت ہے۔
